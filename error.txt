export function num(value, digits = 2) {
  if (value === null || value === undefined || Number.isNaN(Number(value))) {
    return '—'
  }
  const n = Number(value)
  if (n !== 0 && Math.abs(n) < 0.001) {
    return n.toExponential(2)
  }
  return n.toFixed(digits)
}

export function signed(value, digits = 2) {
  if (value === null || value === undefined || Number.isNaN(Number(value))) {
    return '—'
  }
  const n = Number(value)
  const sign = n > 0 ? '+' : ''
  if (n !== 0 && Math.abs(n) < 0.001) {
    return `${sign}${n.toExponential(2)}`
  }
  return `${sign}${n.toFixed(digits)}`
}

export function pval(value) {
  if (value === null || value === undefined || Number.isNaN(Number(value))) {
    return '—'
  }
  const p = Number(value)
  if (p < 0.0001) {
    return '< 0.0001'
  }
  return p.toFixed(4)
}

export function interpret(result) {
  if (!result) return ''
  const layer = result.meta?.layer || 'Selected variable'
  const units = result.meta?.units || ''
  const nObs = result.meta?.n_obs ?? 0
  const agg = result.meta?.aggregation || 'monthly'
  const mk = result.tests?.mann_kendall || {}
  const sen = result.tests?.sen_slope || {}
  const sigLevel = result.significance?.level || 'not evaluated'
  const perDecade = sen.per_decade ?? result.trend?.slope_per_decade

  const direction =
    mk.trend === 'increasing'
      ? 'an increasing trend'
      : mk.trend === 'decreasing'
      ? 'a decreasing trend'
      : 'no monotonic trend'

  return `Across ${nObs} ${agg} observations, ${layer} exhibits ${direction} (${sigLevel}, Mann-Kendall p = ${pval(
    mk.p_value
  )}). The robust Sen's slope rate of change is ${signed(perDecade, 3)} ${units} per decade (95% confidence interval: ${signed(
    sen.ci_low,
    4
  )} to ${signed(sen.ci_high, 4)} ${units}/year).`
}
